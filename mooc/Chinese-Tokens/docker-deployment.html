Docker容器化部署

Docker容器化部署完整指南

🐳 Docker部署优势

  • 环境一致性：开发、测试、生产环境完全一致
  • 快速部署：一键启动，无需配置复杂环境
  • 资源隔离：避免应用间相互影响
  • 弹性扩展：基于容器编排快速应对流量变化

完整的Docker Compose配置

多服务架构设计

# docker-compose.yml
version: '3.8'

services:
  # Nginx反向代理
  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf
      - ./ssl:/etc/nginx/ssl
      - ./logs/nginx:/var/log/nginx
    depends_on:
      - api
    networks:
      - app-network
    restart: unless-stopped

  # 核心API服务
  api:
    build: 
      context: ./api
      dockerfile: Dockerfile
    ports:
      - "3000:3000"
    environment:
      - NODE_ENV=production
      - DATABASE_URL=postgresql://user:password@db:5432/api_gateway
      - REDIS_URL=redis://redis:6379
      - DEEPSEEK_API_KEY=${DEEPSEEK_API_KEY}
      - BAIDU_API_KEY=${BAIDU_API_KEY}
    volumes:
      - ./logs/api:/app/logs
      - ./api/uploads:/app/uploads
    depends_on:
      - db
      - redis
    networks:
      - app-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:3000/health"]
      interval: 30s
      timeout: 10s
      retries: 3

  # PostgreSQL数据库
  db:
    image: postgres:13-alpine
    environment:
      - POSTGRES_DB=api_gateway
      - POSTGRES_USER=user
      - POSTGRES_PASSWORD=password
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./database/init.sql:/docker-entrypoint-initdb.d/init.sql
    networks:
      - app-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U user -d api_gateway"]
      interval: 30s
      timeout: 10s
      retries: 3

  # Redis缓存
  redis:
    image: redis:6-alpine
    command: redis-server --appendonly yes
    volumes:
      - redis_data:/data
    networks:
      - app-network
    restart: unless-stopped
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 30s
      timeout: 10s
      retries: 3

volumes:
  postgres_data:
  redis_data:

networks:
  app-network:
    driver: bridge

Node.js应用Dockerfile

# api/Dockerfile
FROM node:18-alpine AS builder

# 安装系统依赖
RUN apk add --no-cache \
    python3 \
    make \
    g++ \
    curl

# 设置工作目录
WORKDIR /app

# 复制package文件
COPY package*.json ./
COPY tsconfig.json ./

# 安装依赖
RUN npm ci --only=production && npm cache clean --force

# 复制源代码
COPY src/ ./src/
COPY public/ ./public/

# 构建应用
RUN npm run build

# 生产阶段
FROM node:18-alpine

# 创建非root用户（安全最佳实践）
RUN addgroup -g 1001 -S nodejs
RUN adduser -S nextjs -u 1001

# 安装系统依赖
RUN apk add --no-cache curl

# 设置工作目录
WORKDIR /app

# 从构建阶段复制node_modules和构建结果
COPY --from=builder --chown=nextjs:nodejs /app/node_modules ./node_modules
COPY --from=builder --chown=nextjs:nodejs /app/dist ./dist
COPY --from=builder --chown=nextjs:nodejs /app/public ./public

# 复制package.json
COPY --chown=nextjs:nodejs package*.json ./

# 切换用户
USER nextjs

# 暴露端口
EXPOSE 3000

# 健康检查
HEALTHCHECK --interval=30s --timeout=10s --start-period=5s --retries=3 \
  CMD curl -f http://localhost:3000/health || exit 1

# 启动应用
CMD ["node", "dist/server.js"]

部署和运维命令

常用Docker命令

# 构建和启动服务
docker-compose up -d --build

# 查看服务状态
docker-compose ps

# 查看日志
docker-compose logs -f api
docker-compose logs -f nginx

# 停止服务
docker-compose down

# 备份数据库
docker-compose exec db pg_dump -U user api_gateway > backup.sql

# 执行数据库迁移
docker-compose exec api npm run migrate

# 查看资源使用情况
docker stats

# 进入容器调试
docker-compose exec api bash
docker-compose exec db psql -U user api_gateway

环境变量配置 (.env文件)

# 数据库配置
POSTGRES_DB=api_gateway
POSTGRES_USER=user
POSTGRES_PASSWORD=your_secure_password

# Redis配置
REDIS_PASSWORD=your_redis_password

# AI API Keys
DEEPSEEK_API_KEY=your_deepseek_key
BAIDU_API_KEY=your_baidu_key
ALIYUN_API_KEY=your_aliyun_key

# 应用配置
NODE_ENV=production
PORT=3000
BASE_URL=https://yourdomain.com

# 安全配置
JWT_SECRET=your_jwt_secret
ENCRYPTION_KEY=your_encryption_key

# 邮件服务（可选）
SMTP_HOST=smtp.gmail.com
SMTP_PORT=587
SMTP_USER=your_email@gmail.com
SMTP_PASS=your_app_password
