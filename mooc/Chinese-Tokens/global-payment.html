汇率风险管理

实时汇率计算策略

// 汇率管理服务
class ExchangeRateService {
    constructor() {
        this.rates = {};
        this.lastUpdate = null;
        this.updateInterval = 3600000; // 1小时更新一次
    }
    
    async getRate(baseCurrency = 'USD', targetCurrency) {
        // 检查缓存是否有效
        if (this.shouldUpdate()) {
            await this.updateRates();
        }
        
        const rateKey = `${baseCurrency}_${targetCurrency}`;
        if (this.rates[rateKey]) {
            return this.rates[rateKey];
        }
        
        // 如果直接汇率不存在，通过USD中转计算
        if (baseCurrency !== 'USD' && targetCurrency !== 'USD') {
            const usdToBase = this.rates[`USD_${baseCurrency}`];
            const usdToTarget = this.rates[`USD_${targetCurrency}`];
            
            if (usdToBase && usdToTarget) {
                return usdToTarget / usdToBase;
            }
        }
        
        throw new Error(`无法获取汇率: ${baseCurrency} -> ${targetCurrency}`);
    }
    
    async updateRates() {
        try {
            // 从多个汇率API获取数据
            const [provider1, provider2] = await Promise.all([
                this.fetchFromProvider1(),
                this.fetchFromProvider2()
            ]);
            
            // 选择最优汇率
            this.rates = this.mergeRates(provider1, provider2);
            this.lastUpdate = new Date();
            
        } catch (error) {
            console.error('汇率更新失败:', error);
            // 使用缓存数据，如果缓存也没有就抛出错误
            if (!this.lastUpdate) {
                throw error;
            }
        }
    }
    
    shouldUpdate() {
        if (!this.lastUpdate) return true;
        return Date.now() - this.lastUpdate.getTime() > this.updateInterval;
    }
}
