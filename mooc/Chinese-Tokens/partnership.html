佣金计算和结算

// 合作伙伴佣金计算
class PartnerCommission {
    constructor() {
        this.commissionRates = {
            basic: 0.15,    // 15%
            advanced: 0.25, // 25% 
            strategic: 0.30 // 30%
        };
    }
    
    calculateCommission(partner, salesData) {
        const rate = this.commissionRates[partner.tier];
        let commission = 0;
        
        // 计算佣金
        salesData.forEach(sale => {
            const saleCommission = sale.amount * rate;
            commission += saleCommission;
            
            // 记录明细
            this.recordCommissionDetail({
                partnerId: partner.id,
                saleId: sale.id,
                amount: sale.amount,
                rate: rate,
                commission: saleCommission,
                date: sale.date
            });
        });
        
        return commission;
    }
    
    async processPayout(partner, period) {
        const commission = await this.getCommissionForPeriod(partner.id, period);
        
        if (commission.total > partner.minPayoutAmount) {
            // 生成支付记录
            const payout = await this.createPayoutRecord({
                partnerId: partner.id,
                amount: commission.total,
                period: period,
                status: 'pending'
            });
            
            // 调用支付接口
            await this.initiatePayout(payout);
            
            return payout;
        }
        
        return null; // 未达到支付门槛
    }
}
