智能缓存实现

// 缓存管理器实现
class CacheManager {
    constructor() {
        this.redis = new Redis(process.env.REDIS_URL);
        this.memoryCache = new Map();
    }
    
    async getCachedResponse(prompt, model) {
        const cacheKey = this.generateCacheKey(prompt, model);
        
        // 内存缓存检查
        if (this.memoryCache.has(cacheKey)) {
            return this.memoryCache.get(cacheKey);
        }
        
        // Redis缓存检查
        const cached = await this.redis.get(cacheKey);
        if (cached) {
            this.memoryCache.set(cacheKey, cached, 60000);
            return JSON.parse(cached);
        }
        
        return null;
    }
}

// 基于性能的负载均衡
class LoadBalancer {
    async selectProvider(request) {
        const providers = await this.getAvailableProviders();
        const scored = await this.scoreProviders(providers, request);
        return this.selectBestProvider(scored);
    }
    
    async scoreProviders(providers, request) {
        return providers.map(provider => ({
            provider,
            score: this.calculateScore(provider, request)
        })).sort((a, b) => b.score - a.score);
    }
}
