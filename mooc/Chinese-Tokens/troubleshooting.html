监控和诊断工具

系统健康检查脚本

#!/bin/bash
# 系统健康检查脚本

# 检查服务状态
check_service() {
    service=$1
    port=$2
    
    if nc -z localhost $port &>/dev/null; then
        echo "✅ $service 服务正常 (端口: $port)"
        return 0
    else
        echo "❌ $service 服务异常 (端口: $port)"
        return 1
    fi
}

# 检查数据库连接
check_database() {
    if pg_isready -h localhost -p 5432 &>/dev/null; then
        echo "✅ 数据库连接正常"
        return 0
    else
        echo "❌ 数据库连接失败"
        return 1
    fi
}

# 检查磁盘空间
check_disk() {
    usage=$(df / | awk 'NR==2 {print $5}' | sed 's/%//')
    if [ $usage -lt 80 ]; then
        echo "✅ 磁盘空间充足 ($usage%)"
        return 0
    else
        echo "⚠️  磁盘空间紧张 ($usage%)"
        return 1
    fi
}

# 执行检查
echo "=== 系统健康检查 ==="
check_service "Nginx" 80
check_service "API服务" 3000
check_service "Redis" 6379
check_database
check_disk
