API适配层设计与实现

API适配层架构设计

API适配层的核心功能是标准化不同AI厂商的API接口，为前端应用提供统一的OpenAI兼容接口。

🎯 设计目标

  • 100%兼容OpenAI API格式：用户无需修改现有代码
  • 智能路由：自动选择最优的上游服务
  • 故障转移：主服务故障时自动切换
  • 性能监控：实时监控各服务的响应时间

请求适配器核心实现

OpenAI适配器类

// openai-adapter.js
export class OpenAIAdapter {
    static async handleOpenAIRequest(openaiRequest, userApiKey) {
        try {
            // 1. 验证API密钥
            const user = await AuthService.authenticate(userApiKey);
            if (!user) {
                throw new Error('Invalid API key');
            }
            
            // 2. 检查使用额度和频率限制
            await RateLimitService.checkLimit(user.id, openaiRequest);
            
            // 3. 验证请求参数
            this.validateRequest(openaiRequest);
            
            // 4. 选择最优的上游服务
            const providerConfig = await this.selectBestProvider(openaiRequest, user);
            
            // 5. 转换请求格式
            const providerRequest = this.convertToProviderFormat(openaiRequest, providerConfig);
            
            // 6. 发送请求到上游服务
            const providerResponse = await this.sendToProvider(providerRequest, providerConfig);
            
            // 7. 转换响应格式
            const openaiResponse = this.convertToOpenAIFormat(providerResponse, providerConfig);
            
            // 8. 记录使用量
            await UsageService.recordUsage(user.id, openaiRequest, openaiResponse, providerConfig);
            
            return openaiResponse;
            
        } catch (error) {
            // 错误处理和故障转移
            return await this.handleError(error, openaiRequest, userApiKey);
        }
    }
    
    static async selectBestProvider(openaiRequest, user) {
        const availableProviders = await ProviderHealthService.getHealthyProviders();
        
        // 基于性能、成本、用户偏好选择
        const scoredProviders = await Promise.all(
            availableProviders.map(async (provider) => ({
                provider,
                score: await this.calculateProviderScore(provider, openaiRequest, user)
            }))
        );
        
        // 选择分数最高的提供商
        const bestProvider = scoredProviders.reduce((best, current) => 
            current.score > best.score ? current : best
        );
        
        return bestProvider.provider;
    }
    
    static convertToProviderFormat(openaiRequest, providerConfig) {
        const baseRequest = {
            model: providerConfig.model,
            messages: openaiRequest.messages,
            max_tokens: openaiRequest.max_tokens || 2048,
            temperature: openaiRequest.temperature || 0.7,
            top_p: openaiRequest.top_p || 1.0
        };
        
        // 厂商特定的参数转换
        switch (providerConfig.provider) {
            case 'deepseek':
                return {
                    ...baseRequest,
                    stream: openaiRequest.stream || false,
                    presence_penalty: openaiRequest.presence_penalty,
                    frequency_penalty: openaiRequest.frequency_penalty
                };
                
            case 'baidu':
                return {
                    messages: openaiRequest.messages.map(msg => ({
                        role: msg.role,
                        content: msg.content
                    })),
                    stream: false, // 百度暂不支持流式响应
                    temperature: openaiRequest.temperature
                };
                
            case 'alibaba':
                return {
                    model: providerConfig.model,
                    input: {
                        messages: openaiRequest.messages
                    },
                    parameters: {
                        max_tokens: openaiRequest.max_tokens,
                        temperature: openaiRequest.temperature
                    }
                };
                
            default:
                return baseRequest;
        }
    }
}
